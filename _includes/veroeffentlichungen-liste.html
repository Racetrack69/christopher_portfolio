{%- comment -%} Eine Liste von Veröffentlichungen. Parameter: daten (Einträge aus _data/veroeffentlichungen.yml). Nach Jahr gruppiert, neueste oben, innerhalb eines Jahres in der Reihenfolge der Datei. Das Jahr steht nur beim ersten Eintrag eines Jahres. Einträge ohne Jahr folgen am Ende. Hat kein Eintrag ein Jahr, entfällt die Jahresspalte; hat ein Eintrag ein Cover, kommt eine Spalte für die Titelseiten dazu. Leere Angaben werden nicht angezeigt. Links auf fremde Seiten öffnen in einem neuen Fenster (mit Hinweis für Screenreader). {%- endcomment -%} {%- assign daten = include.daten -%} {%- assign mit_jahr = daten | where_exp: "v", "v.jahr != nil" -%} {%- assign ohne_jahr = daten | where_exp: "v", "v.jahr == nil" -%} {%- assign mit_cover = daten | where_exp: "v", "v.cover != nil" -%} {%- assign gruppen = mit_jahr | group_by: "jahr" | sort: "name" | reverse -%} {%- assign liste = "" | split: "" -%} {%- for g in gruppen -%}{%- assign liste = liste | concat: g.items -%}{%- endfor -%} {%- assign liste = liste | concat: ohne_jahr -%} {%- assign vorher = "" -%}
    {%- for v in liste %} {%- assign link = v.link | strip -%} {%- assign ausgabe = v.ausgabe | strip -%} {%- assign titel = v.titel | strip -%} {%- assign inhalt = v.inhalt | strip -%} {%- assign jahr = v.jahr | append: "" -%} {%- capture kopf -%}{% if titel != "" %}{{ titel }}{% else %}{{ v.magazin }}{% endif %}{%- endcapture -%} {%- capture det -%} {%- if titel != "" %}{{ v.magazin }}{% endif -%} {%- if titel != "" and ausgabe != "" %} · {% endif %}{{ ausgabe -}} {%- if inhalt != "" %}{% if titel != "" or ausgabe != "" %} · {% endif %}{{ inhalt }}{% endif -%} {%- endcapture -%} {%- assign serie = nil -%} {%- if v.serie -%} {%- assign serie_url = "/serien/" | append: v.serie | append: "/" -%} {%- assign serie = site.serien | where: "url", serie_url | first -%} {%- endif %}
  1. {%- if mit_jahr.size > 0 %} {% if jahr != vorher %}{{ jahr }}{% endif %} {%- endif %} {%- if mit_cover.size > 0 %} {% if v.cover %}{% endif %} {%- endif %}
    {% if link != "" %}{{ kopf }} (öffnet in neuem Fenster){% else %}{{ kopf }}{% endif %}
    {%- if det != "" %}
    {{ det }}
    {%- endif %} {%- if link != "" or serie %}
    {%- if link != "" %}{% if v.online %}Online lesen{% else %}Zum Heft{% endif %} ↗ (öffnet in neuem Fenster){% endif -%} {%- if serie %}Zur Serie {{ serie.title }} →{% endif -%}
    {%- endif %}
    2. {%- assign vorher = jahr -%} {%- endfor %}
